```typescript
// Enterprise knowledge integration
class EnterpriseKnowledgeHub {
  async processDocument(document: Document) {
    // Extract entities and relationships
    const entities = await this.mcp.simpleMemory.createEntities([
      { name: document.title, type: 'document', observations: [document.summary] },
    ]);

    // Store in persistent memory with context
    await this.mcp.memorai.remember(document.content, {
      entityType: 'enterprise_document',
      project: document.project,
      priority: 'high',
    });

    // Analyze for insights
    const insights = await this.mcp.sequentialThinking.sequentialthinking({
      thought: `Analyze document: ${document.title}`,
      thoughtNumber: 1,
      totalThoughts: 3,
      nextThoughtNeeded: true,
    });

    return { entities, insights };
  }
}
```
