Real-time Memory Updates
WebSocket integration for live memory updates
Real-time WebSocket
class RealtimeMemorySubscription { constructor(config) { this.sdk = new MemorAISDK(config); this.isSubscribed = false; this.listeners = new Map(); } // Subscribe to memory updates async subscribe(filters = {}) { try { await this.sdk.connect(); // Subscribe to memory events this.sdk.on('memoryCreated', (memory) => { this.notifyListeners('created', memory); }); this.sdk.on('memoryUpdated', (memory) => { this.notifyListeners('updated', memory); }); this.sdk.on('memoryDeleted', (memoryId) => { this.notifyListeners('deleted', { id: memoryId }); }); // Apply subscription filters if (Object.keys(filters).length > 0) { await this.sdk.setSubscriptionFilters(filters); } this.isSubscribed = true; console.log('Subscribed to memory updates'); } catch (error) { console.error('Failed to subscribe:', error); throw error; } } // Add event listener addEventListener(event, callback) { if (!this.listeners.has(event)) { this.listeners.set(event, new Set()); } this.listeners.get(event).add(callback); } // Remove event listener removeEventListener(event, callback) { const listeners = this.listeners.get(event); if (listeners) { listeners.delete(callback); } } // Notify all listeners of an event notifyListeners(event, data) { const listeners = this.listeners.get(event); if (listeners) { listeners.forEach(callback => { try { callback(data); } catch (error) { console.error('Listener error:', error); } }); } } // Unsubscribe async unsubscribe() { try { await this.sdk.disconnect(); this.isSubscribed = false; this.listeners.clear(); console.log('Unsubscribed from memory updates'); } catch (error) { console.error('Failed to unsubscribe:', error); } } } // Usage example const subscription = new RealtimeMemorySubscription({ baseURL: 'http://localhost:4950', apiKey: 'your-api-key', enableWebSocket: true }); // Add event listeners subscription.addEventListener('created', (memory) => { console.log('New memory created:', memory.content); updateUI('add', memory); }); subscription.addEventListener('updated', (memory) => { console.log('Memory updated:', memory.content); updateUI('update', memory); }); subscription.addEventListener('deleted', (memory) => { console.log('Memory deleted:', memory.id); updateUI('remove', memory); }); // Start subscription with filters subscription.subscribe({ importance: { min: 5 }, tags: ['important', 'urgent'], agentId: 'my-agent' }); function updateUI(action, memory) { // Update your UI based on the memory event const memoryElement = document.getElementById(`memory-${memory.id}`); switch (action) { case 'add': addMemoryToUI(memory); break; case 'update': updateMemoryInUI(memory); break; case 'remove': if (memoryElement) { memoryElement.remove(); } break; } } // Cleanup when page unloads window.addEventListener('beforeunload', () => { subscription.unsubscribe(); });

AI-Powered Memory Clustering
Advanced clustering with semantic analysis and quality metrics
Advanced AI
async function performAdvancedClustering(sdk, agentId) { try { // Configure clustering options const clusteringOptions = { // Semantic weight - how much to consider content similarity semanticWeight: 0.7, // Temporal weight - how much to consider time-based grouping temporalWeight: 0.2, // Importance weight - how much to consider importance scores importanceWeight: 0.1, // Minimum cluster size minClusterSize: 3, // Maximum number of clusters maxClusters: 10, // Quality threshold - minimum acceptable silhouette score qualityThreshold: 0.3, // Enable advanced features enableSemanticEnhancement: true, includeKeywordExtraction: true, includeSentimentAnalysis: true }; // Perform clustering console.log('Starting memory clustering...'); const clusterResult = await sdk.clusterMemories({ agentId: agentId, clusterCount: 5, // Desired number of clusters options: clusteringOptions }); // Analyze clustering results const analysis = analyzeClusters(clusterResult); // Generate cluster reports const reports = generateClusterReports(clusterResult); // Optimize clustering if quality is low if (clusterResult.silhouetteScore < clusteringOptions.qualityThreshold) { console.log('Clustering quality is low, attempting optimization...'); const optimizedResult=await optimizeClustering(sdk, agentId, clusterResult); return optimizedResult; } return { clustering: clusterResult, analysis: analysis, reports: reports }; } catch (error) { console.error('Clustering failed:', error); throw error; } } function analyzeClusters(clusterResult) { const analysis={ totalMemories: 0, clusterSizes: [], qualityMetrics: { overallScore: clusterResult.silhouetteScore, cohesionScores: [], separationScores: [] }, topKeywords: new Set(), themesIdentified: [] }; clusterResult.clusters.forEach((cluster, index)=> { analysis.totalMemories += cluster.memories.length; analysis.clusterSizes.push(cluster.memories.length); analysis.qualityMetrics.cohesionScores.push(cluster.cohesionScore); analysis.qualityMetrics.separationScores.push(cluster.separationScore); // Collect keywords cluster.keywords.forEach(keyword => analysis.topKeywords.add(keyword)); // Identify themes if (cluster.theme) { analysis.themesIdentified.push({ clusterId: index, theme: cluster.theme, confidence: cluster.themeConfidence, memoryCount: cluster.memories.length }); } }); // Calculate statistics analysis.averageClusterSize = analysis.totalMemories / clusterResult.clusters.length; analysis.clusterSizeVariance = calculateVariance(analysis.clusterSizes); analysis.topKeywords = Array.from(analysis.topKeywords).slice(0, 20); return analysis; } function generateClusterReports(clusterResult) { return clusterResult.clusters.map((cluster, index) => { // Calculate cluster statistics const memoryCount = cluster.memories.length; const avgImportance = cluster.memories.reduce((sum, mem) => sum + (mem.metadata?.importance || 0), 0) / memoryCount; const dateRange = getDateRange(cluster.memories); // Extract common patterns const commonPatterns = extractCommonPatterns(cluster.memories); // Generate summary const summary = generateClusterSummary(cluster); return { clusterId: index, title: `Cluster ${index + 1}: ${cluster.theme || 'Mixed Content'}`, summary: summary, statistics: { memoryCount: memoryCount, averageImportance: avgImportance, dateRange: dateRange, qualityScore: cluster.qualityScore, cohesionScore: cluster.cohesionScore, separationScore: cluster.separationScore }, keywords: cluster.keywords, commonPatterns: commonPatterns, recommendations: generateClusterRecommendations(cluster) }; }); } async function optimizeClustering(sdk, agentId, initialResult) { const optimizationStrategies = [ // Try different cluster counts { clusterCount: Math.max(2, initialResult.clusters.length - 1) }, { clusterCount: Math.min(10, initialResult.clusters.length + 1) }, // Adjust weight distributions { clusterCount: initialResult.clusters.length, options: { semanticWeight: 0.8, temporalWeight: 0.15, importanceWeight: 0.05 } }, { clusterCount: initialResult.clusters.length, options: { semanticWeight: 0.6, temporalWeight: 0.3, importanceWeight: 0.1 } } ]; let bestResult = initialResult; let bestScore = initialResult.silhouetteScore; for (const strategy of optimizationStrategies) { try { const result = await sdk.clusterMemories({ agentId: agentId, clusterCount: strategy.clusterCount, options: { enableSemanticEnhancement: true, includeKeywordExtraction: true, ...strategy.options } }); if (result.silhouetteScore > bestScore) { bestResult = result; bestScore = result.silhouetteScore; console.log(`Improved clustering score: ${bestScore}`); } } catch (error) { console.warn('Optimization strategy failed:', error); } } return bestResult; } function calculateVariance(numbers) { const mean = numbers.reduce((sum, n) => sum + n, 0) / numbers.length; const squaredDiffs = numbers.map(n => Math.pow(n - mean, 2)); return squaredDiffs.reduce((sum, sq) => sum + sq, 0) / numbers.length; } function getDateRange(memories) { const dates = memories.map(m => new Date(m.createdAt)); return { earliest: new Date(Math.min(...dates)), latest: new Date(Math.max(...dates)) }; } function extractCommonPatterns(memories) { const patterns = { commonTags: {}, frequentPhrases: {}, importanceLevels: {} }; memories.forEach(memory => { // Count common tags if (memory.metadata?.tags) { memory.metadata.tags.forEach(tag => { patterns.commonTags[tag] = (patterns.commonTags[tag] || 0) + 1; }); } // Count importance levels const importance = memory.metadata?.importance || 0; const level = importance < 3 ? 'low' : importance < 7 ? 'medium' : 'high' ; patterns.importanceLevels[level]=(patterns.importanceLevels[level] || 0) + 1; }); return patterns; } function generateClusterSummary(cluster) { const memoryCount=cluster.memories.length; const theme=cluster.theme || 'mixed topics' ; const avgImportance=cluster.memories.reduce((sum, mem)=> sum + (mem.metadata?.importance || 0), 0) / memoryCount; return `This cluster contains ${memoryCount} memories related to ${theme}. ` + `The average importance score is ${avgImportance.toFixed(1)}, ` + `and the cluster quality score is ${cluster.qualityScore.toFixed(3)}.`; } function generateClusterRecommendations(cluster) { const recommendations = []; if (cluster.qualityScore < 0.3) { recommendations.push('Consider splitting this cluster for better cohesion'); } if (cluster.memories.length < 3) { recommendations.push('This cluster may be too small; consider merging with similar clusters'); } if (cluster.separationScore < 0.5) { recommendations.push('Increase semantic differentiation from other clusters'); } const avgImportance=cluster.memories.reduce((sum, mem)=> sum + (mem.metadata?.importance || 0), 0) / cluster.memories.length; if (avgImportance > 7) { recommendations.push('High-importance cluster; consider prioritizing in UI'); } else if (avgImportance < 3) { recommendations.push('Low-importance cluster; consider archiving old memories'); } return recommendations; } // Usage example async function runClusteringExample() { const sdk=new MemorAISDK({ baseURL: 'http://localhost:4950' , apiKey: 'your-api-key' }); try { const result=await performAdvancedClustering(sdk, 'data-analysis-agent' ); console.log('Clustering Results:'); console.log(`Total clusters: ${result.clustering.clusters.length}`); console.log(`Overall quality score: ${result.clustering.silhouetteScore}`); console.log(`Total memories clustered: ${result.analysis.totalMemories}`); // Display cluster reports result.reports.forEach((report, index)=> { console.log(`\n${report.title}`); console.log(report.summary); console.log('Keywords:', report.keywords.join(', ')); console.log('Recommendations:', report.recommendations); }); } catch (error) { console.error('Clustering example failed:', error); } } // Run the example runClusteringExample();
