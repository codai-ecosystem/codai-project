Integrate MemorAI with Angular applications using services and dependency injection

📦 Installation

npm install @memorai/sdk @memorai/angular # or ng add @memorai/angular

⚙️ Module Setup

// app.module.ts import { NgModule } from '@angular/core'; import { MemorAIModule } from '@memorai/angular'; @NgModule({ imports: [ MemorAIModule.forRoot({ baseURL: environment.memoraiBaseURL, apiKey: environment.memoraiApiKey, defaultAgentId: 'angular-app' }) ] }) export class AppModule { }

Environment Configuration

// environment.ts export const environment = { memoraiBaseURL: 'http://localhost:4950', memoraiApiKey: 'your-api-key', memoraiAgentId: 'angular-application' };

🔧 Service Integration

// memory.service.ts import { Injectable } from '@angular/core'; import { MemorAIService } from '@memorai/angular'; import { Observable } from 'rxjs'; @Injectable({ providedIn: 'root' }) export class MemoryService { constructor(private memorai: MemorAIService) {} async storeMemory(content: string, metadata?: any) { return this.memorai.remember({ content, agentId: 'angular-app', metadata: { importance: 7, ...metadata } }); } async searchMemories(query: string): Promise { return this.memorai.recall({ query, agentId: 'angular-app', limit: 20 }); } getConnectionStatus(): Observable { return this.memorai.connectionStatus$; } }

📱 Component Example

// memory-manager.component.ts import { Component, OnInit } from '@angular/core'; import { MemoryService } from './memory.service'; @Component({ selector: 'app-memory-manager', template: `
{{ connectionStatus ? 'Connected' : 'Disconnected' }}
{{ memory.content }} ★{{ memory.metadata?.importance }}
` }) export class MemoryManagerComponent implements OnInit { newContent = ''; searchQuery = ''; memories: any[] = []; connectionStatus = false; constructor(private memoryService: MemoryService) {} ngOnInit() { this.memoryService.getConnectionStatus() .subscribe(status => this.connectionStatus = status); } async addMemory() { if (!this.newContent.trim()) return; try { await this.memoryService.storeMemory(this.newContent, { timestamp: new Date().toISOString() }); this.newContent = ''; } catch (error) { console.error('Failed to add memory:', error); } } async search() { if (!this.searchQuery.trim()) return; try { this.memories = await this.memoryService.searchMemories(this.searchQuery); } catch (error) { console.error('Search failed:', error); } } get statusClass() { return this.connectionStatus ? 'connected' : 'disconnected'; } }

🔍 Advanced Usage

// Advanced service with RxJS import { BehaviorSubject, Observable, from } from 'rxjs'; import { map, catchError } from 'rxjs/operators'; @Injectable() export class AdvancedMemoryService { private memories$ = new BehaviorSubject([]); constructor(private memorai: MemorAIService) {} getMemories(): Observable { return this.memories$.asObservable(); } searchMemoriesRx(query: string): Observable { return from(this.memorai.recall({ query, limit: 50 })).pipe( map(results => { this.memories$.next(results); return results; }), catchError(error => { console.error('Search error:', error); return []; }) ); } }
