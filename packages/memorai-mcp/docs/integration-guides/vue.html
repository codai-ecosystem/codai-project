📦 Installation

Install MemorAI SDK with Vue 3 integration support. The package includes TypeScript definitions and Vue 3 composables.

# Install MemorAI SDK with Vue integration npm install @memorai/sdk @memorai/vue # Or with Yarn yarn add @memorai/sdk @memorai/vue # Or with PNPM pnpm add @memorai/sdk @memorai/vue

⚙️ Setup & Configuration

Configure MemorAI in your Vue 3 application with the plugin system or direct composable usage.

Global Plugin Setup (Recommended)

// main.js or main.ts import { createApp } from 'vue' import { MemorAIPlugin } from '@memorai/vue' import App from './App.vue' const app = createApp(App) // Configure MemorAI plugin app.use(MemorAIPlugin, { baseURL: process.env.VUE_APP_MEMORAI_BASE_URL || 'http://localhost:4950', apiKey: process.env.VUE_APP_MEMORAI_API_KEY, defaultAgentId: 'vue-app', enableWebSocket: true, retryConfig: { maxRetries: 3, retryDelay: 1000, backoffMultiplier: 2 }, // Global error handler onError: (error) => { console.error('MemorAI Error:', error) } }) app.mount('#app')

Environment Variables Setup

# .env.local VUE_APP_MEMORAI_BASE_URL=http://localhost:4950 VUE_APP_MEMORAI_API_KEY=your-api-key-here VUE_APP_MEMORAI_AGENT_ID=vue-application

Manual Configuration (Alternative)

// composables/memorai.js import { MemorAISDK } from '@memorai/sdk' export const memoraiSDK = new MemorAISDK({ baseURL: process.env.VUE_APP_MEMORAI_BASE_URL, apiKey: process.env.VUE_APP_MEMORAI_API_KEY, enableWebSocket: true, onError: (error) => { console.error('MemorAI SDK Error:', error) } }) export { useMemorAI, useMemorySearch, useMemoryAnalytics } from '@memorai/vue'

// Basic usage in a component import { useMemorAI } from '@memorai/vue' export default { setup() { const { remember, recall, updateMemory, forgetMemory, isConnected, isLoading, error } = useMemorAI({ agentId: 'user-preferences' }) const storeUserPreference = async (preference) => { try { const memory = await remember({ content: `User prefers ${preference}`, agentId: 'user-preferences', metadata: { importance: 7, tags: ['preferences', 'ui'], category: 'user-settings' } }) console.log('Preference stored:', memory) } catch (err) { console.error('Failed to store preference:', err) } } const searchPreferences = async (query) => { try { const results = await recall({ query, agentId: 'user-preferences', limit: 10 }) return results } catch (err) { console.error('Search failed:', err) return [] } } return { storeUserPreference, searchPreferences, isConnected, isLoading, error } } }
useMemorySearch - Advanced Search Functionality
const { search, results, isSearching, searchError, filters, updateFilters, clearResults } = useMemorySearch(config?)
import { useMemorySearch } from '@memorai/vue' import { ref, computed } from 'vue' export default { setup() { const { search, results, isSearching, searchError, filters, updateFilters, clearResults } = useMemorySearch({ agentId: 'content-manager', defaultFilters: { minImportance: 5, tags: ['important'] } }) const searchQuery = ref('') const performSearch = async () => { if (!searchQuery.value.trim()) return await search({ query: searchQuery.value, limit: 20 }) } const filteredResults = computed(() => { return results.value.filter(memory => memory.metadata?.importance >= filters.value.minImportance ) }) const updateImportanceFilter = (importance) => { updateFilters({ minImportance: importance }) } return { searchQuery, performSearch, filteredResults, isSearching, searchError, updateImportanceFilter, clearResults } } }
useMemoryAnalytics - Analytics & Insights
const { analytics, insights, clusters, refreshAnalytics, isAnalyzing, analyticsError } = useMemoryAnalytics(agentId, options?)
import { useMemoryAnalytics } from '@memorai/vue' import { onMounted, watch } from 'vue' export default { setup() { const { analytics, insights, clusters, refreshAnalytics, isAnalyzing, analyticsError } = useMemoryAnalytics('analytics-agent', { autoRefresh: true, refreshInterval: 30000, // 30 seconds includeInsights: true, includeClusters: true }) // Auto-refresh on mount onMounted(() => { refreshAnalytics() }) // Watch for changes and log watch([analytics, insights, clusters], ([newAnalytics, newInsights, newClusters]) => { if (newAnalytics) { console.log('Analytics updated:', newAnalytics) } if (newInsights) { console.log('Insights generated:', newInsights) } if (newClusters) { console.log('Clusters updated:', newClusters) } }) return { analytics, insights, clusters, refreshAnalytics, isAnalyzing, analyticsError } } }

📘 TypeScript Integration

MemorAI provides excellent TypeScript support with full type definitions for Vue 3 composables.

// types/memorai.d.ts - Custom type extensions import { Ref } from 'vue' export interface UserMemoryData { userId: string preferences: Record settings: UserSettings lastLogin: Date } export interface UserSettings { theme: 'light' | 'dark' | 'auto' language: string notifications: boolean privacy: PrivacySettings } export interface PrivacySettings { shareAnalytics: boolean shareUsage: boolean allowCookies: boolean } // Composable with TypeScript export interface MemoryManagerState { memories: Ref isLoading: Ref error: Ref searchQuery: Ref filters: Ref } // Component with TypeScript export default defineComponent({ name: 'TypedMemoryComponent', setup(): MemoryManagerState { const { remember, recall, isConnected, isLoading, error } = useMemorAI({ agentId: 'user-manager', enableWebSocket: true }) const memories: Ref[]> = ref([]) const searchQuery: Ref = ref('') const filters: Ref = ref({ minImportance: 5, tags: ['important'], dateRange: { start: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000), end: new Date() } }) const storeUserMemory = async (userData: UserMemoryData): Promise> => { return await remember({ content: `User ${userData.userId} preferences updated`, agentId: 'user-manager', metadata: { importance: 8, tags: ['user-data', 'preferences'], entityType: 'user_memory', priority: 'high' }, data: userData }) } const searchUserMemories = async (userId: string): Promise[]> => { const results = await recall({ query: `user ${userId}`, agentId: 'user-manager', limit: 50, includeOtherAgents: false }) // Type assertion with runtime validation return results.filter((memory): memory is Memory => { return memory.data && 'userId' in memory.data }) } return { memories, isLoading, error, searchQuery, filters, storeUserMemory, searchUserMemories } } })

TypeScript Best Practices

  • Use generic types for memory data: useMemorAI<MyDataType>()
  • Define interfaces for memory metadata and custom data
  • Use strict TypeScript configuration for better type safety
  • Leverage Vue 3's defineComponent for better type inference
  • Create custom composables with proper return type annotations
  • Use type guards for runtime validation of memory data

✨ Best Practices

Follow these best practices to build robust, performant Vue 3 applications with MemorAI.

Performance Optimization

  • Use shallowRef for large memory collections to reduce reactivity overhead
  • Implement virtual scrolling for large memory lists
  • Debounce search inputs to prevent excessive API calls
  • Cache frequently accessed memories using Vue's computed properties
  • Use readonly refs for data that shouldn't be mutated
  • Implement proper cleanup in onUnmounted hooks
// Performance optimized memory search import { ref, shallowRef, computed, watch, onUnmounted } from 'vue' import { debounce } from 'lodash-es' export default { setup() { const { search, isSearching } = useMemorySearch() // Use shallowRef for large collections const memories = shallowRef([]) const searchQuery = ref('') // Debounced search function const debouncedSearch = debounce(async (query) => { if (!query.trim()) return const results = await search({ query, limit: 100 }) // Replace entire array for shallowRef memories.value = results }, 300) // Watch search query with debouncing watch(searchQuery, (newQuery) => { debouncedSearch(newQuery) }) // Computed filtered results const filteredMemories = computed(() => { return memories.value.filter(memory => memory.metadata?.importance >= 5 ) }) // Cleanup onUnmounted(() => { debouncedSearch.cancel() }) return { searchQuery, filteredMemories, isSearching } } }

Error Handling & Reliability

  • Always handle errors gracefully with try-catch blocks
  • Provide user-friendly error messages and recovery options
  • Implement retry logic for failed operations
  • Use loading states to improve user experience
  • Validate data before sending to MemorAI APIs
  • Implement offline fallbacks when possible

Code Organization

  • Create custom composables for reusable MemorAI logic
  • Use provide/inject for sharing MemorAI configuration
  • Separate business logic from UI components
  • Use TypeScript for better type safety and developer experience
  • Follow Vue 3 composition API best practices
  • Document your memory schemas and agent configurations

🔧 Troubleshooting

Common issues and solutions when integrating MemorAI with Vue 3.

Common Issues

Connection Issues

// Debug connection problems import { useMemorAI } from '@memorai/vue' import { onMounted } from 'vue' export default { setup() { const { isConnected, error } = useMemorAI({ baseURL: process.env.VUE_APP_MEMORAI_BASE_URL, timeout: 10000, // 10 second timeout retryConfig: { maxRetries: 3, retryDelay: 2000 }, onError: (error) => { console.error('MemorAI Connection Error:', error) if (error.code === 'ECONNREFUSED') { console.log('Is MemorAI server running?') } if (error.status === 401) { console.log('Check your API key configuration') } if (error.status === 429) { console.log('Rate limit exceeded, implementing backoff...') } } }) onMounted(() => { // Test connection on mount if (!isConnected.value) { console.warn('Not connected to MemorAI server') } }) return { isConnected, error } } }

Reactivity Issues

// Fix reactivity problems import { ref, reactive, toRefs } from 'vue' // ❌ Don't destructure reactive objects directly const { remember, recall } = useMemorAI() // Loses reactivity // ✅ Use toRefs for destructuring const memoraiState = useMemorAI() const { remember, recall, isConnected, error } = toRefs(memoraiState) // ✅ Or access properties directly const memorai = useMemorAI() // memorai.isConnected, memorai.error, etc.

Memory Updates Not Reflecting

// Ensure proper state updates const { recall } = useMemorAI() const memories = ref([]) const updateMemoryList = async () => { try { const results = await recall({ query: 'all', limit: 100 }) // ✅ Replace entire array to trigger reactivity memories.value = results // ❌ Don't mutate existing array elements // memories.value.push(...results) // May not trigger updates } catch (error) { console.error('Failed to update memories:', error) } }

Debug Tips

  • Use Vue DevTools to inspect reactive state
  • Enable MemorAI debug logging in development
  • Check network tab for API request/response details
  • Verify environment variables are properly loaded
  • Test with different agent IDs to isolate issues
  • Use browser console to inspect MemorAI SDK state
